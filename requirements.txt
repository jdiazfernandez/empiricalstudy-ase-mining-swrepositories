# langdetect backs the corpus-language analysis (src/qual/detect_language.py).
#
# KNOWN INSTALL FAILURE, and why the code has a fallback: langdetect 1.0.9 ships
# only as a source distribution, so pip must build it. On a Windows account whose
# user name contains a "$", setuptools reads the "$..." as a variable while
# computing install paths and aborts with
#     ValueError: invalid variable '<...>'
# The build failure is entirely about the account name; the library itself is
# pure Python and runs fine. On such a machine, unpack the sdist anywhere and
# point the code at it instead of installing:
#
#     LANGDETECT_PATH=/path/to/unpacked  python src/qual/h5_language_sensitivity.py
#
# Everywhere else the line below installs normally and no variable is needed.
langdetect==1.0.9
